**Accept the bounded allocation approach, subject to the named integration decisions below.**

### 5.3 Read-Only Expansion Recipe

Run this block from repository root after extracting it from this report, without writing a
script or manifest elsewhere. It checks the fixed register hash, hashes required files only,
never follows a symlink, and emits allocation metadata only. Its output is the exact expanded
manifest: each slice inherits its file's recorded SHA-256. Labels identify primary owners.
No reading status is changed by expansion. No code, package, fixture or generated JS executes.
The recipe's `primary` list identifies exactly all required paths and hashes; `packets` gives
every inclusive text scope and reading-credit subdivision; `structured` gives exact JSON
pointers or safe-disposition scope. Combining them is the explicit coverage manifest.
The existing validator does not ingest this embedded selector language. The coordinator must
expand it into ordinary assignments/receipts; no tooling change is proposed or required.

```javascript
const fs = require('node:fs');
const crypto = require('node:crypto');
const assert = require('node:assert/strict');
const reportPath = "kupua/exploration/docs/03 Ce n'est pas une pipe dream/media-api-work/api-boundary-09-p06-corpus-allocation.md";
const report = fs.readFileSync(reportPath, 'utf8');
const manifest = JSON.parse(report.match(/```json\n([\s\S]*?)\n```/)[1]);
const digest = value => crypto.createHash('sha256').update(value).digest('hex');
const registerBytes = fs.readFileSync(manifest.inventory);
assert.equal(digest(registerBytes), manifest.inventorySha256);
const register = JSON.parse(registerBytes);
const required = register.files.filter(entry => entry.disposition === 'review-required');
assert.equal(required.length, manifest.requiredCount);
assert.equal(new Set(required.map(entry => entry.path)).size, required.length);
const primary = [];
const units = [];
const structured = [];
const compare = (left, right) => left < right ? -1 : left > right ? 1 : 0;
const pointerEscape = value => value.replaceAll('~', '~0').replaceAll('/', '~1');
const leaves = (value, pointer = '') => {
  if (value === null || typeof value === 'object' === false || Object.keys(value).length === 0) return [pointer];
  return Object.entries(value).flatMap(([key, child]) => leaves(child, pointer + '/' + pointerEscape(key)));
};
const openPointers = (value, pointer, credited) => {
  if (credited.includes(pointer)) return [];
  if (credited.some(scope => scope.startsWith(pointer + '/')) === false) return [pointer];
  return Object.entries(value).flatMap(([key, child]) => openPointers(child, pointer + '/' + pointerEscape(key), credited));
};
for (const entry of [...required].sort((left, right) => compare(left.path, right.path))) {
  const stat = fs.lstatSync(entry.path);
  assert.equal(stat.isSymbolicLink(), false, 'Symlink requires coordinator decision: ' + entry.path);
  assert.equal(stat.isFile(), true);
  const bytes = fs.readFileSync(entry.path);
  assert.equal(digest(bytes), entry.sha256, 'Changed input: ' + entry.path);
  assert.equal(bytes.length, entry.bytes);
  const credited = manifest.creditedFull[entry.path];
  const family = credited ?? manifest.rules.find(rule => new RegExp(rule.pattern, 'i').test(entry.path)).owner;
  primary.push({path: entry.path, sha256: entry.sha256, family, lines: entry.lines, bytes: entry.bytes, preservedReceipts: entry.receipts});
  if (credited) continue;
  if (family === 'CANON') {
    const data = JSON.parse(bytes);
    const rootPointers = Object.keys(data).filter(key => (key === 'entries') === false).map(key => '/' + pointerEscape(key));
    const groups = [];
    let group = {pointers: rootPointers, bytes: rootPointers.reduce((sum, pointer) => sum + Buffer.byteLength(JSON.stringify(data[pointer.slice(1)])), 0), entries: 0};
    for (const [index, value] of data.entries.entries()) {
      const size = Buffer.byteLength(JSON.stringify(value));
      assert.ok(size <= 24000, 'Split oversized entry by metric pointers before dispatch');
      if (group.entries === 4 || group.bytes + size > 24000) {
        groups.push(group);
        group = {pointers: [], bytes: 0, entries: 0};
      }
      group.pointers.push('/entries/' + index);
      group.bytes += size;
      group.entries++;
    }
    if (group.pointers.length > 0) groups.push(group);
    const credits = [...new Set(entry.receipts.filter(receipt => receipt.sha256 === entry.sha256 && receipt.method === 'json-pointers').flatMap(receipt => receipt.pointers))];
    const resolve = pointer => pointer.split('/').slice(1).reduce((value, key) => value[key.replaceAll('~1', '/').replaceAll('~0', '~')], data);
    for (const scope of credits) assert.ok(resolve(scope) === undefined ? false : true);
    const owned = [];
    groups.forEach((value, index) => {
      const creditedPointers = credits.filter(scope => value.pointers.some(pointer => scope === pointer || scope.startsWith(pointer + '/')));
      const unreadPointers = value.pointers.flatMap(pointer => openPointers(resolve(pointer), pointer, creditedPointers));
      owned.push(...creditedPointers, ...unreadPointers);
```
